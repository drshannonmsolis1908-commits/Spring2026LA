\left[\begin{array}{ccccc}
    1 & -7 & 0 & 6 & 5  \\ 
    0 & 0 & 1 & -2 & -3 \\ 
    -1 & 7 & -4 & 2 & 7 
\end{array}\right]
      
\end{problem}
